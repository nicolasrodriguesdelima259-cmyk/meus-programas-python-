def enviar_mensagem(grupo_concelho_dos_cirilo, mensagem, limite):

    if 'driver' not in globals() or driver is None:
        print("Erro: O WebDriver (driver) não foi inicializado. Descomente e execute a linha de inicialização do driver antes de chamar esta função.")
        return

    try:
        grupo = driver.find_element(By.XPATH, f'//span[@title="{grupo_concelho_dos_cirilo}"]')
        grupo.click()

        time.sleep(2)

        for i in range(limite):
            campo_mensagem = driver.find_element(By.XPATH, '//div[@contenteditable="true"][@data-tab="1"]')
            campo_mensagem.send_keys(mensagem)
            campo_mensagem.send_keys("\n")
            print(f'Mensagem {i + 1} enviada!')
            time.sleep(3)

    except Exception as e:
        print(f"Erro ao enviar mensagem: {e}")
